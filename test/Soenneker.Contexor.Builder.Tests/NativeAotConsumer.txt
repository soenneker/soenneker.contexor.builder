if (JsonSerializer.IsReflectionEnabledByDefault) throw new Exception("JSON reflection must be disabled.");
using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(45));
var ct = timeout.Token;
var context = RpcJsonContext.Default;
var input = JsonSerializer.Deserialize("""{"type":"text","text":"native","text_elements":[]}""", context.UserInput)!;
if (!JsonSerializer.Serialize(input, context.UserInput).Contains("native")) throw new Exception("Union roundtrip failed.");
var parameters = new ThreadStartParams { Model = new Optional<string?>(null), ApprovalsReviewer = new Optional<ApprovalsReviewer?>(null) };
string serialized = JsonSerializer.Serialize(parameters, context.ThreadStartParams);
if (!serialized.Contains("\"model\":null") || serialized.Contains("\"cwd\"")) throw new Exception("Optional semantics failed.");
var restored = JsonSerializer.Deserialize(serialized, context.ThreadStartParams)!;
if (!restored.ApprovalsReviewer.IsDefined || restored.ApprovalsReviewer.Value != null) throw new Exception("Nullable enum conversion failed.");

await using (var transport = new ProbeTransport())
await using (var client = new CodexClient(transport))
{
    var listed = await client.ThreadLoadedList(new ThreadLoadedListParams(), ct);
    if (listed.Data[0] != "native-thread") throw new Exception("Typed RPC failed.");
    var received = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
    client.ThreadArchivedReceived += notification => { if (notification.ThreadId == "native-thread") received.TrySetResult(); };
    transport.Emit("""{"method":"thread/archived","params":{"threadId":"native-thread"}}""");
    await received.Task.WaitAsync(ct);
    client.AttestationGenerateHandler = (_, _) => ValueTask.FromResult(new AttestationGenerateResponse { Token = "native-token" });
    transport.Emit("""{"method":"attestation/generate","id":7,"params":{}}""");
    JsonElement reply = await transport.Reply.Task.WaitAsync(ct);
    if (reply.GetProperty("id").GetInt32() != 7 || reply.GetProperty("result").GetProperty("token").GetString() != "native-token") throw new Exception("Typed server handler failed.");
}

if (Environment.GetEnvironmentVariable("CONTEXOR_CODEX_EXECUTABLE") is { Length: > 0 } executable)

sealed class ProbeTransport : IJsonRpcTransport
{
    private readonly Channel<JsonElement> _incoming = Channel.CreateUnbounded<JsonElement>();
    public Task Completion => _incoming.Reader.Completion;
    internal readonly TaskCompletionSource<JsonElement> Reply = new(TaskCreationOptions.RunContinuationsAsynchronously);
    internal void Emit(string json) { using var doc = JsonDocument.Parse(json); _incoming.Writer.TryWrite(doc.RootElement.Clone()); }
    public ValueTask<JsonElement?> Send(JsonElement request, bool notification, CancellationToken cancellationToken = default)
    {
        if (notification) return ValueTask.FromResult<JsonElement?>(null);
        using var doc = JsonDocument.Parse("{\"id\":" + request.GetProperty("id").GetRawText() + ",\"result\":{\"data\":[\"native-thread\"],\"nextCursor\":null}}");
        return ValueTask.FromResult<JsonElement?>(doc.RootElement.Clone());
    }
    public async IAsyncEnumerable<JsonElement> ReadMessages([EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await foreach (var message in _incoming.Reader.ReadAllAsync(cancellationToken)) yield return message;
    }
    public ValueTask WriteResponse(JsonElement response, CancellationToken cancellationToken = default) { Reply.TrySetResult(response); return ValueTask.CompletedTask; }
    public ValueTask DisposeAsync() { _incoming.Writer.TryComplete(); return ValueTask.CompletedTask; }
}
