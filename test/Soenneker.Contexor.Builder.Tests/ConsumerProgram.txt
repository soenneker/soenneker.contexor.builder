static void Check(bool condition, string message) { if (!condition) throw new Exception(message); }
static async Task Reject<T>(Func<Task> action) where T : Exception
{
    try { await action(); } catch (T) { return; }
    throw new Exception("Expected " + typeof(T).Name);
}
string Name(string wire) => string.Concat(wire.Split(new[] { '/', '.', '_', '-' }).Select(s => char.ToUpperInvariant(s[0]) + s[1..]));
JsonNode? Sample(JsonNode? node, int depth = 0)
{
    if (depth > 60) throw new Exception("Recursive required schema: " + node);
    if (node is not JsonObject schema) return JsonValue.Create("arbitrary-json");
    if (schema["$ref"] is { } reference) return Sample(definitions[reference.ToString().Split('/').Last()], depth + 1);
    if (schema["enum"] is JsonArray values) return values[0]?.DeepClone();
    if (schema["const"] is { } constant) return constant.DeepClone();
    foreach (string key in new[] { "oneOf", "anyOf", "allOf" })
        if (schema[key] is JsonArray branches)
        {
            JsonNode? chosen = Sample(branches.FirstOrDefault(b => b?["type"]?.ToString() == "null") ?? branches[0], depth + 1);
            if (chosen is JsonObject combined && schema["required"] is JsonArray common)
                foreach (var member in common) combined[member!.ToString()] = Sample(schema["properties"]![member.ToString()], depth + 1);
            return chosen;
        }
    if (schema["type"] is JsonArray types)
    {
        if (types.Any(t => t?.ToString() == "null")) return null;
        schema = schema.DeepClone().AsObject(); schema["type"] = types[0]!.DeepClone();
    }
    switch (schema["type"]?.ToString())
    {
        case "null": return null;
        case "object":
            var result = new JsonObject();
            foreach (var member in (schema["required"] as JsonArray ?? []))
                result[member!.ToString()] = Sample(schema["properties"]![member.ToString()], depth + 1);
            return result;
        case "array": return new JsonArray();
        case "boolean": return JsonValue.Create(false);
        case "integer": return JsonValue.Create((long)Math.Ceiling(schema["minimum"]?.GetValue<decimal>() ?? 1));
        case "number": return schema["minimum"]?.DeepClone() ?? JsonValue.Create(1);
        case "string": return JsonValue.Create(schema["format"]?.ToString() == "uri" ? "https://example.com" : "sample");
        default: return JsonValue.Create("arbitrary-json");
    }
}
string ResultName(string wire, JsonNode? parameters)
{
    if (overrides.TryGetValue(wire, out var mapped)) return mapped;
    string? reference = parameters?["$ref"]?.ToString().Split('/').Last();
    return reference?.EndsWith("Params", StringComparison.Ordinal) == true ? reference[..^6] + "Response" : Name(wire) + "Response";
}
async Task<JsonObject> Read(StreamReader reader) => JsonNode.Parse(await reader.ReadLineAsync(ct) ?? throw new EndOfStreamException())!.AsObject();
async Task Write(StreamWriter writer, JsonObject message) => await writer.WriteLineAsync(message.ToJsonString().AsMemory(), ct);
JsonObject Reply(JsonObject request, JsonNode? result) => new() { ["id"] = request["id"]!.DeepClone(), ["result"] = result };

// Exercise every generated outgoing request over a real duplex stream.
int calls = 0;
var ids = new HashSet<string>();
foreach (JsonNode envelope in definitions["ClientRequest"]!["oneOf"]!.AsArray()!)
{
    string wire = envelope["properties"]!["method"]!["enum"]![0]!.ToString();
    MethodInfo method = typeof(CodexClient).GetMethod(Name(wire))!;
    var parameters = method.GetParameters();
    JsonNode? parameterSchema = envelope["properties"]!["params"];
    JsonNode? payload = parameterSchema == null ? null : Sample(parameterSchema);
    object?[] arguments = parameters.Length == 1 ? [ct] : [JsonSerializer.Deserialize(payload?.ToJsonString() ?? "null", parameters[0].ParameterType, json), ct];
    object operation = method.Invoke(client, arguments)!;
    Task task = (Task)operation.GetType().GetMethod("AsTask")!.Invoke(operation, null)!;
    JsonObject sent = await Read(peerReader);
    Check(!sent.ContainsKey("jsonrpc") && sent["method"]!.ToString() == wire, "Codex wire method/version: " + wire);
    Check(ids.Add(sent["id"]!.ToString()), "Unique ID");
    Check(JsonNode.DeepEquals(sent["params"], payload), "Exact typed parameter roundtrip: " + wire);
    string responseName = ResultName(wire, parameterSchema);
    JsonNode? response = Sample(definitions[responseName]);
    await Write(peerWriter, Reply(sent, response));
    try { await task.WaitAsync(ct); }
    catch (Exception e) { throw new Exception("Response failed for " + wire + ": " + response?.ToJsonString(), e); }
    object? typed = task.GetType().GetProperty("Result")!.GetValue(task);
    Check(JsonNode.DeepEquals(JsonNode.Parse(JsonSerializer.Serialize(typed, typed!.GetType(), json)), response), "Exact typed result roundtrip: " + wire);
    calls++;
}
Check(calls == 85, "All 85 requests were exercised");

var initialized = await Read(peerReader);
Check(initialized["method"]!.ToString() == "initialized" && !initialized.ContainsKey("id"), "initialized notification");

// Every declared server notification dispatches its typed payload.
int notificationCount = 0;
foreach (JsonNode envelope in definitions["ServerNotification"]!["oneOf"]!.AsArray()!)
{
    string wire = envelope["properties"]!["method"]!["enum"]![0]!.ToString();
    EventInfo ev = typeof(CodexClient).GetEvent(Name(wire) + "Received")!;
    var received = new TaskCompletionSource<object>(TaskCreationOptions.RunContinuationsAsynchronously);
    Type type = ev.EventHandlerType!.GenericTypeArguments[0];
    var parameter = Expression.Parameter(type);
    var callback = Expression.Lambda(ev.EventHandlerType, Expression.Call(Expression.Constant(received), typeof(TaskCompletionSource<object>).GetMethod("SetResult")!, Expression.Convert(parameter, typeof(object))), parameter).Compile();
    ev.AddEventHandler(client, callback);
    JsonNode? payload = Sample(envelope["properties"]!["params"]);
    await Write(peerWriter, new JsonObject { ["method"] = wire, ["params"] = payload });
    object value = await received.Task.WaitAsync(ct);
    Check(JsonNode.DeepEquals(JsonNode.Parse(JsonSerializer.Serialize(value, type, json)), payload), "Notification roundtrip: " + wire);
    ev.RemoveEventHandler(client, callback);
    notificationCount++;
}
Check(notificationCount == 66, "All 66 notifications dispatched");

// Every server-request handler is typed and replies with the peer's numeric ID.
int serverRequests = 0;
foreach (JsonNode envelope in definitions["ServerRequest"]!["oneOf"]!.AsArray()!)
{
    string wire = envelope["properties"]!["method"]!["enum"]![0]!.ToString();
    PropertyInfo property = typeof(CodexClient).GetProperty(Name(wire) + "Handler")!;
    Type[] generic = property.PropertyType.GenericTypeArguments;
    JsonNode? response = Sample(definitions[ResultName(wire, envelope["properties"]!["params"])]);
    Type responseType = generic[2].GenericTypeArguments[0];
    Delegate callback = (Delegate)typeof(Handlers).GetMethod("Create")!.MakeGenericMethod(generic[0], responseType).Invoke(null, [response?.ToJsonString() ?? "null"])!;
    property.SetValue(client, callback);
    await Write(peerWriter, new JsonObject { ["method"] = wire, ["id"] = 100 + serverRequests, ["params"] = Sample(envelope["properties"]!["params"]) });
    JsonObject reply = await Read(peerReader);
    Check(reply["id"]!.GetValue<int>() == 100 + serverRequests && JsonNode.DeepEquals(reply["result"], response), "Server request reply: " + wire + " " + reply);
    property.SetValue(client, null);
    serverRequests++;
}

// Roundtrip every branch of every named union, not just the first branch used by method samples.
int unionBranches = 0;
foreach (var definition in definitions)
{
    if (definition.Value is not JsonObject schema) continue;
    var branches = schema["oneOf"] as JsonArray ?? schema["anyOf"] as JsonArray;
    Type? type = typeof(CodexClient).Assembly.GetType("Example.Models." + Name(definition.Key));
    if (branches is null || type?.GetProperty("Variant") is null) continue;
    foreach (var branch in branches)
    {
        var sample = Sample(branch);
        if (sample is JsonObject obj && schema["required"] is JsonArray common)
            foreach (var member in common) obj[member!.ToString()] = Sample(schema["properties"]![member.ToString()]);
        object? value;
        try { value = JsonSerializer.Deserialize(sample?.ToJsonString() ?? "null", type, json); }
        catch (Exception e) { throw new Exception("Union branch " + definition.Key + ": " + sample, e); }
        Check(JsonNode.DeepEquals(JsonNode.Parse(JsonSerializer.Serialize(value, type, json)), sample), "Union branch roundtrip: " + definition.Key);
        unionBranches++;
    }
}
Check(unionBranches > 100, "Comprehensive union branch coverage");

static class Handlers
{
    public static Func<T, CancellationToken, ValueTask<R>> Create<T, R>(string result) => (parameters, ct) => ValueTask.FromResult(JsonSerializer.Deserialize<R>(result, RpcJsonContext.Default.Options)!);
}
